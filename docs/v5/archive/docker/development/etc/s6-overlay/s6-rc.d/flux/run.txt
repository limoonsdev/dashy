#!/bin/sh

cd /var/www/html

. /etc/s6-overlay/scripts/container-role
role="$(dashy_container_role_value)"

if ! dashy_container_has_role flux; then
    echo "   INFO  Flux is disabled for role '$role', sleeping."
    exec sleep infinity
fi

if grep -qE '^DASHY_FLUX_ENABLED=false' .env 2>/dev/null || [ "${DASHY_FLUX_ENABLED:-}" = "false" ]; then
    echo "   INFO  Flux is disabled, sleeping."
    exec sleep infinity
fi

export DASHY_FLUX_GRPC_BIND="${DASHY_FLUX_GRPC_BIND:-0.0.0.0:6443}"
export DASHY_FLUX_UNIX_SOCKET_PATH="${DASHY_FLUX_UNIX_SOCKET_PATH:-/run/dashy/flux.sock}"
export DASHY_FLUX_JWT_PRIVATE_KEY_PATH="${DASHY_FLUX_JWT_PRIVATE_KEY_PATH:-/var/www/html/storage/app/flux/jwt.priv}"
export DASHY_FLUX_JWT_PUBLIC_KEY_PATH="${DASHY_FLUX_JWT_PUBLIC_KEY_PATH:-/var/www/html/storage/app/flux/jwt.pub}"
export DASHY_FLUX_ALLOW_PUBLIC_BIND="${DASHY_FLUX_ALLOW_PUBLIC_BIND:-1}"
export DASHY_FLUX_REQUIRE_HOST_BINDING="${DASHY_FLUX_REQUIRE_HOST_BINDING:-0}"
export DASHY_FLUX_LARAVEL_API_URL="${DASHY_FLUX_LARAVEL_API_URL:-http://127.0.0.1:8080}"
if [ -z "${DASHY_FLUX_LARAVEL_API_TOKEN:-}" ] && [ -f .env ]; then
    DASHY_FLUX_LARAVEL_API_TOKEN="$(grep -E '^DASHY_FLUX_LARAVEL_API_TOKEN=' .env 2>/dev/null | tail -n1 | cut -d= -f2- | sed "s/^['\"]//; s/['\"]$//")"
fi
export DASHY_FLUX_LARAVEL_API_TOKEN="${DASHY_FLUX_LARAVEL_API_TOKEN:-}"
if [ ! -r "$DASHY_FLUX_JWT_PUBLIC_KEY_PATH" ]; then
    echo "   INFO  Flux JWT public key not found at $DASHY_FLUX_JWT_PUBLIC_KEY_PATH, generating keypair..."
    mkdir -p "$(dirname "$DASHY_FLUX_JWT_PRIVATE_KEY_PATH")" "$(dirname "$DASHY_FLUX_JWT_PUBLIC_KEY_PATH")"
    openssl genpkey -algorithm EC -pkeyopt ec_paramgen_curve:P-256 -out "$DASHY_FLUX_JWT_PRIVATE_KEY_PATH.tmp"
    chmod 0600 "$DASHY_FLUX_JWT_PRIVATE_KEY_PATH.tmp"
    mv "$DASHY_FLUX_JWT_PRIVATE_KEY_PATH.tmp" "$DASHY_FLUX_JWT_PRIVATE_KEY_PATH"
    openssl pkey -in "$DASHY_FLUX_JWT_PRIVATE_KEY_PATH" -pubout -out "$DASHY_FLUX_JWT_PUBLIC_KEY_PATH.tmp"
    chmod 0644 "$DASHY_FLUX_JWT_PUBLIC_KEY_PATH.tmp"
    mv "$DASHY_FLUX_JWT_PUBLIC_KEY_PATH.tmp" "$DASHY_FLUX_JWT_PUBLIC_KEY_PATH"
fi

if ! /usr/local/bin/flux --version >/dev/null 2>&1; then
    echo "   ERROR  Flux binary cannot run in this container. Check that the installed coold nightly Flux artifact is compatible with Alpine."
    exec sleep infinity
fi

mkdir -p "$(dirname "$DASHY_FLUX_UNIX_SOCKET_PATH")"

echo "   INFO  Flux is enabled for role '$role', starting..."
exec /usr/local/bin/flux
